import { CommonModule } from '@angular/common';
import { Component, OnInit } from '@angular/core';
import { ActivatedRoute, Router } from '@angular/router';
import { FormBuilder, FormGroup, ReactiveFormsModule, Validators } from '@angular/forms';
import {
  DxButtonModule,
  DxDateBoxModule,
  DxNumberBoxModule,
  DxPopupModule,
  DxSelectBoxModule,
  DxTabPanelModule,
  DxTextAreaModule,
  DxTextBoxModule,
} from 'devextreme-angular';
import { finalize } from 'rxjs';
import { CommonGridComponent } from '../../../../layout/common-grid/common-grid.component';
import { AuthService } from '../../../../services/auth.service';
import { ChipNavigationService } from '../../../../services/ChipNavigation.service';
import { ToastService } from '../../../../services/toaster.service';
import {
  AddDeduct,
  AddRecoveryModel,
  CasePriority,
  CheckKind,
  ClearanceDepartment,
  DuesType,
  FnfCaseModel,
  FnfCheckModel,
  FnfRecoveryModel,
  FnfSettlementDetailModel,
  LookupOption,
  RecoveryStatus,
  SettlementComponentType,
} from '../../models/fnf.model';
import { FnfApiService } from '../../services/fnf-api.service';

const CASE_PRIORITIES: CasePriority[] = ['Normal', 'Management Separation'];
const CLEARANCE_DEPARTMENTS: ClearanceDepartment[] = ['IT', 'Admin', 'Finance', 'Manager', 'Other'];
const DUES_TYPES: DuesType[] = ['Asset Not Returned', 'Asset Damage', 'Advance', 'Loan', 'Other'];
const RECOVERY_STATUSES: RecoveryStatus[] = ['Confirmed', 'Waived'];
const COMPONENT_TYPES: SettlementComponentType[] = [
  'FinalSalary',
  'LeaveEncashment',
  'Gratuity',
  'NoticePay',
  'NoticeRecovery',
  'LoanRecovery',
  'AssetRecovery',
  'OtherRecovery',
];
const ADD_DEDUCT: AddDeduct[] = ['Add', 'Deduct'];

/** Preparer works through clearance/recovery/settlement/verification, then submits. */
@Component({
  selector: 'app-fnf-settlement-case',
  standalone: true,
  imports: [
    CommonModule,
    ReactiveFormsModule,
    CommonGridComponent,
    DxTabPanelModule,
    DxPopupModule,
    DxSelectBoxModule,
    DxDateBoxModule,
    DxNumberBoxModule,
    DxTextAreaModule,
    DxTextBoxModule,
    DxButtonModule,
  ],
  templateUrl: './case.component.html',
})
export class FnfSettlementCaseComponent implements OnInit {
  caseId = 0;
  case: FnfCaseModel | null = null;
  loading = false;
  saving = false;
  recalculating = false;
  rechecking = false;
  submitting = false;

  casePriorities = CASE_PRIORITIES;
  logins: LookupOption[] = [];
  salaryHeads: LookupOption[] = [];

  planningForm: FormGroup;

  readinessChecks: FnfCheckModel[] = [];
  verificationChecks: FnfCheckModel[] = [];
  handoverChecks: FnfCheckModel[] = [];
  readinessColumns: Record<string, unknown>[] = [];
  verificationColumns: Record<string, unknown>[] = [];
  handoverColumns: Record<string, unknown>[] = [];

  recoveries: FnfRecoveryModel[] = [];
  recoveryColumns: Record<string, unknown>[] = [];

  settlementDetails: FnfSettlementDetailModel[] = [];
  settlementColumns: Record<string, unknown>[] = [];

  // ------------------------------------------------------------- Check popup
  checkPopupVisible = false;
  checkBusy = false;
  private checkPopupKind: CheckKind = 'Readiness';
  checkPopupRow: FnfCheckModel | null = null;
  checkResultOptions: string[] = [];
  checkResult: string | null = null;
  checkRemarks = '';

  // ---------------------------------------------------------- Recovery popup
  recoveryPopupVisible = false;
  recoveryBusy = false;
  isRecoveryEdit = false;
  clearanceDepartments = CLEARANCE_DEPARTMENTS;
  duesTypes = DUES_TYPES;
  recoveryStatuses = RECOVERY_STATUSES;
  recoveryForm: FormGroup;
  private editingRecoveryId: number | null = null;

  // ------------------------------------------------------------- Detail popup
  detailPopupVisible = false;
  detailBusy = false;
  isDetailEdit = false;
  componentTypes = COMPONENT_TYPES;
  addDeductOptions = ADD_DEDUCT;
  detailForm: FormGroup;
  private editingDetailId: number | null = null;

  constructor(
    private authService: AuthService,
    private fnfApi: FnfApiService,
    private toast: ToastService,
    private route: ActivatedRoute,
    private router: Router,
    private fb: FormBuilder,
    private chipService: ChipNavigationService,
  ) {
    this.authService.ScreenName = 'F&F Case';

    this.planningForm = this.fb.group({
      casePriority: ['Normal', Validators.required],
      plannedCalcStartDate: [null],
      plannedSettlementDate: [null],
      actualCalcStartDate: [null],
      approvalTurnaroundDays: [null],
      hrExecutiveUserMasterId: [null],
      payrollExecutiveUserMasterId: [null],
      financeExecutiveUserMasterId: [null],
      approverUserMasterId: [null],
      remarks: [''],
    });

    this.recoveryForm = this.fb.group({
      clearanceDepartment: ['IT', Validators.required],
      duesType: ['Advance', Validators.required],
      description: [''],
      amount: [0, Validators.required],
      recoveryStatus: ['Confirmed', Validators.required],
      remarks: [''],
    });

    this.detailForm = this.fb.group({
      componentType: ['OtherRecovery', Validators.required],
      salaryHeadMasterId: [null],
      addDeduct: ['Deduct', Validators.required],
      amount: [0, Validators.required],
      remarks: [''],
    });
  }

  ngOnInit(): void {
    this.caseId = Number(this.route.snapshot.paramMap.get('id') ?? 0);
    this.chipService.addTab({ title: 'F&F Case', route: `/fnf/settlement/case/${this.caseId}` });

    this.readinessColumns = this.buildCheckColumns();
    this.verificationColumns = this.buildCheckColumns();
    this.handoverColumns = this.buildCheckColumns();

    this.recoveryColumns = [
      { dataField: 'clearanceDepartment', caption: 'Department', width: 110 },
      { dataField: 'duesType', caption: 'Dues Type', width: 150 },
      { dataField: 'description', caption: 'Description' },
      { dataField: 'amount', caption: 'Amount', dataType: 'number', format: { type: 'fixedPoint', precision: 2 }, width: 120 },
      { dataField: 'recoveryStatus', caption: 'Status', width: 110 },
      { dataField: 'remarks', caption: 'Remarks' },
    ];

    this.settlementColumns = [
      { dataField: 'componentType', caption: 'Component' },
      { dataField: 'addDeduct', caption: 'Add/Deduct', width: 100 },
      { dataField: 'days', caption: 'Days', dataType: 'number', width: 80 },
      { dataField: 'rate', caption: 'Rate', dataType: 'number', format: { type: 'fixedPoint', precision: 2 }, width: 100 },
      { dataField: 'amount', caption: 'Amount', dataType: 'number', format: { type: 'fixedPoint', precision: 2 }, width: 120 },
      {
        dataField: 'isAuto',
        caption: 'Source',
        width: 90,
        cellTemplate: (container: HTMLElement, options: any) => {
          container.innerText = options.data?.isAuto ? 'Auto' : 'Manual';
        },
      },
      { dataField: 'remarks', caption: 'Remarks / Basis' },
    ];

    this.fnfApi.getLogins().subscribe({ next: (list) => (this.logins = list) });
    this.fnfApi.getSalaryHeads().subscribe({ next: (list) => (this.salaryHeads = list) });

    this.load();
  }

  private buildCheckColumns(): Record<string, unknown>[] {
    return [
      { dataField: 'checkName', caption: 'Requirement' },
      {
        dataField: 'isRequired',
        caption: 'Required',
        width: 90,
        cellTemplate: (container: HTMLElement, options: any) => {
          container.innerText = options.data?.isRequired ? 'Yes' : 'No';
        },
      },
      {
        dataField: 'result',
        caption: 'Result',
        width: 130,
        cellTemplate: (container: HTMLElement, options: any) => {
          const span = document.createElement('span');
          const result = options.data?.result;
          const cls = result === 'Yes' || result === 'Pass' ? 'status-approved' : result === 'No' || result === 'Fail' ? 'status-rejected' : 'status-hold';
          span.className = `status-pill ${cls}`;
          span.innerText = result || 'Not answered';
          container.appendChild(span);
        },
      },
      { dataField: 'policyStandard', caption: 'Policy / Standard' },
      { dataField: 'actualValue', caption: 'Actual', width: 100 },
      { dataField: 'remarks', caption: 'Remarks' },
    ];
  }

  get canEdit(): boolean {
    return this.case?.workflowStatus === 'Draft';
  }

  load(): void {
    if (!this.caseId) return;
    this.loading = true;
    this.fnfApi
      .getCase(this.caseId)
      .pipe(finalize(() => (this.loading = false)))
      .subscribe({
        next: (res) => {
          if (!FnfApiService.isOk(res)) {
            this.toast.error(FnfApiService.messageOf(res, 'Failed to load case.'));
            return;
          }
          this.case = FnfApiService.dataOf<FnfCaseModel>(res);
          this.planningForm.reset(this.case);
        },
        error: (err) => this.toast.error(FnfApiService.errorMessage(err, 'Failed to load case.')),
      });

    this.loadChecks('Readiness');
    this.loadChecks('Verification');
    this.loadChecks('Handover');
    this.loadRecoveries();
    this.loadSettlementDetails();
  }

  private loadChecks(kind: CheckKind): void {
    this.fnfApi.getChecks(this.caseId, kind).subscribe({
      next: (res) => {
        const rows = FnfApiService.dataOf<FnfCheckModel[]>(res) ?? [];
        const sorted = Array.isArray(rows) ? [...rows].sort((a, b) => a.sortOrder - b.sortOrder) : [];
        if (kind === 'Readiness') this.readinessChecks = sorted;
        else if (kind === 'Verification') this.verificationChecks = sorted;
        else this.handoverChecks = sorted;
      },
    });
  }

  private loadRecoveries(): void {
    this.fnfApi.getRecoveries(this.caseId).subscribe({
      next: (res) => {
        const rows = FnfApiService.dataOf<FnfRecoveryModel[]>(res);
        this.recoveries = Array.isArray(rows) ? rows : [];
      },
    });
  }

  private loadSettlementDetails(): void {
    this.fnfApi.getSettlementDetails(this.caseId).subscribe({
      next: (res) => {
        const rows = FnfApiService.dataOf<FnfSettlementDetailModel[]>(res);
        this.settlementDetails = Array.isArray(rows) ? rows : [];
      },
    });
  }

  goBack(): void {
    this.router.navigate(['/fnf/settlement']);
  }

  // -------------------------------------------------------------- Planning

  savePlanning(): void {
    if (!this.case) return;
    this.saving = true;
    this.fnfApi
      .savePlanning({ fnfSettlementId: this.caseId, ...this.planningForm.value })
      .pipe(finalize(() => (this.saving = false)))
      .subscribe({
        next: (res) => {
          if (!FnfApiService.isOk(res)) {
            this.toast.error(FnfApiService.messageOf(res, 'Save failed.'));
            return;
          }
          this.toast.success(FnfApiService.messageOf(res, 'Saved.'));
          this.load();
        },
        error: (err) => this.toast.error(FnfApiService.errorMessage(err, 'Unable to save.')),
      });
  }

  recheck(): void {
    this.rechecking = true;
    this.fnfApi
      .recheck(this.caseId)
      .pipe(finalize(() => (this.rechecking = false)))
      .subscribe({
        next: (res) => {
          this.toast[FnfApiService.isOk(res) ? 'success' : 'error'](FnfApiService.messageOf(res, 'Rechecked.'));
          this.load();
        },
        error: (err) => this.toast.error(FnfApiService.errorMessage(err, 'Unable to recheck.')),
      });
  }

  calculate(): void {
    if (!this.case?.isClearanceReady) {
      this.toast.error(this.case?.clearanceBlockReason || 'Clearance is not ready yet.');
      return;
    }
    this.recalculating = true;
    this.fnfApi
      .calculate(this.caseId)
      .pipe(finalize(() => (this.recalculating = false)))
      .subscribe({
        next: (res) => {
          if (!FnfApiService.isOk(res)) {
            this.toast.error(FnfApiService.messageOf(res, 'Calculation failed.'));
            return;
          }
          this.toast.success(FnfApiService.messageOf(res, 'Calculated.'));
          this.load();
        },
        error: (err) => this.toast.error(FnfApiService.errorMessage(err, 'Unable to calculate.')),
      });
  }

  submit(): void {
    if (!confirm('Submit this settlement for approval? It cannot be edited afterwards.')) return;
    this.submitting = true;
    this.fnfApi
      .submit(this.caseId)
      .pipe(finalize(() => (this.submitting = false)))
      .subscribe({
        next: (res) => {
          if (!FnfApiService.isOk(res)) {
            this.toast.error(FnfApiService.messageOf(res, 'Submit failed.'));
            return;
          }
          this.toast.success(FnfApiService.messageOf(res, 'Submitted for approval.'));
          this.load();
        },
        error: (err) => this.toast.error(FnfApiService.errorMessage(err, 'Unable to submit.')),
      });
  }

  // ---------------------------------------------------------------- Checks

  openCheckEdit(kind: CheckKind, row: FnfCheckModel): void {
    this.checkPopupKind = kind;
    this.checkPopupRow = row;
    this.checkResult = row.result;
    this.checkRemarks = row.remarks ?? '';
    this.checkResultOptions = kind === 'Readiness' || kind === 'Handover' ? ['Yes', 'No'] : ['Pass', 'Fail', 'NA'];
    this.checkPopupVisible = true;
  }

  saveCheck(): void {
    if (!this.checkPopupRow) return;
    this.checkBusy = true;
    this.fnfApi
      .updateCheck(this.caseId, { fnfCheckId: this.checkPopupRow.fnfCheckId, result: this.checkResult, remarks: this.checkRemarks })
      .pipe(finalize(() => (this.checkBusy = false)))
      .subscribe({
        next: (res) => {
          if (!FnfApiService.isOk(res)) {
            this.toast.error(FnfApiService.messageOf(res, 'Could not update.'));
            return;
          }
          this.checkPopupVisible = false;
          this.loadChecks(this.checkPopupKind);
          if (this.checkPopupKind === 'Readiness') this.load();
        },
        error: (err) => this.toast.error(FnfApiService.errorMessage(err, 'Could not update.')),
      });
  }

  // -------------------------------------------------------------- Recovery

  onAddRecovery(): void {
    this.isRecoveryEdit = false;
    this.editingRecoveryId = null;
    this.recoveryForm.reset({
      clearanceDepartment: 'IT',
      duesType: 'Advance',
      description: '',
      amount: 0,
      recoveryStatus: 'Confirmed',
      remarks: '',
    });
    this.recoveryPopupVisible = true;
  }

  onEditRecovery(row: FnfRecoveryModel): void {
    this.isRecoveryEdit = true;
    this.editingRecoveryId = row.fnfRecoveryId;
    this.recoveryForm.reset({
      clearanceDepartment: row.clearanceDepartment,
      duesType: row.duesType,
      description: row.description ?? '',
      amount: row.amount,
      recoveryStatus: row.recoveryStatus,
      remarks: row.remarks ?? '',
    });
    this.recoveryPopupVisible = true;
  }

  onDeleteRecovery(row: FnfRecoveryModel): void {
    this.fnfApi.deleteRecovery(this.caseId, row.fnfRecoveryId).subscribe({
      next: (res) => {
        if (!FnfApiService.isOk(res)) {
          this.toast.error(FnfApiService.messageOf(res, 'Delete failed.'));
          return;
        }
        this.toast.success('Recovery removed.');
        this.loadRecoveries();
      },
      error: (err) => this.toast.error(FnfApiService.errorMessage(err, 'Unable to delete recovery.')),
    });
  }

  saveRecovery(): void {
    if (this.recoveryForm.invalid) {
      this.recoveryForm.markAllAsTouched();
      return;
    }
    this.recoveryBusy = true;
    const value: AddRecoveryModel = { fnfSettlementId: this.caseId, ...this.recoveryForm.value };
    const request$ = this.isRecoveryEdit
      ? this.fnfApi.updateRecovery(this.caseId, { fnfRecoveryId: this.editingRecoveryId!, ...value })
      : this.fnfApi.addRecovery(value);

    request$.pipe(finalize(() => (this.recoveryBusy = false))).subscribe({
      next: (res) => {
        if (!FnfApiService.isOk(res)) {
          this.toast.error(FnfApiService.messageOf(res, 'Save failed.'));
          return;
        }
        this.toast.success(FnfApiService.messageOf(res, 'Saved.'));
        this.recoveryPopupVisible = false;
        this.loadRecoveries();
      },
      error: (err) => this.toast.error(FnfApiService.errorMessage(err, 'Unable to save recovery.')),
    });
  }

  get recoveryTotal(): number {
    return this.recoveries
      .filter((r) => r.recoveryStatus === 'Confirmed' || r.recoveryStatus === 'Waived')
      .reduce((sum, r) => sum + (r.amount || 0), 0);
  }

  // -------------------------------------------------------- Settlement detail

  onAddDetail(): void {
    this.isDetailEdit = false;
    this.editingDetailId = null;
    this.detailForm.reset({ componentType: 'OtherRecovery', salaryHeadMasterId: null, addDeduct: 'Deduct', amount: 0, remarks: '' });
    this.detailPopupVisible = true;
  }

  onEditDetail(row: FnfSettlementDetailModel): void {
    if (row.isAuto) {
      this.toast.error('This line is calculated automatically — edit the configuration or recalculate instead.');
      return;
    }
    this.isDetailEdit = true;
    this.editingDetailId = row.fnfSettlementDetailId;
    this.detailForm.reset({
      componentType: row.componentType,
      salaryHeadMasterId: row.salaryHeadMasterId,
      addDeduct: row.addDeduct,
      amount: row.amount,
      remarks: row.remarks ?? '',
    });
    this.detailPopupVisible = true;
  }

  onDeleteDetail(row: FnfSettlementDetailModel): void {
    if (row.isAuto) {
      this.toast.error('This line is calculated automatically — it would come back on the next Calculate. Remove its cause instead (for example waive the recovery).');
      return;
    }
    this.fnfApi.deleteManualDetail(this.caseId, row.fnfSettlementDetailId).subscribe({
      next: (res) => {
        if (!FnfApiService.isOk(res)) {
          this.toast.error(FnfApiService.messageOf(res, 'Delete failed.'));
          return;
        }
        this.toast.success('Line removed.');
        this.loadSettlementDetails();
      },
      error: (err) => this.toast.error(FnfApiService.errorMessage(err, 'Unable to delete line.')),
    });
  }

  saveDetail(): void {
    if (this.detailForm.invalid) {
      this.detailForm.markAllAsTouched();
      return;
    }
    this.detailBusy = true;
    const value = this.detailForm.value;
    const request$ = this.isDetailEdit
      ? this.fnfApi.updateManualDetail(this.caseId, { fnfSettlementDetailId: this.editingDetailId!, ...value })
      : this.fnfApi.addManualDetail({ fnfSettlementId: this.caseId, ...value });

    request$.pipe(finalize(() => (this.detailBusy = false))).subscribe({
      next: (res) => {
        if (!FnfApiService.isOk(res)) {
          this.toast.error(FnfApiService.messageOf(res, 'Save failed.'));
          return;
        }
        this.toast.success(FnfApiService.messageOf(res, 'Saved.'));
        this.detailPopupVisible = false;
        this.loadSettlementDetails();
        this.load();
      },
      error: (err) => this.toast.error(FnfApiService.errorMessage(err, 'Unable to save line.')),
    });
  }
}
