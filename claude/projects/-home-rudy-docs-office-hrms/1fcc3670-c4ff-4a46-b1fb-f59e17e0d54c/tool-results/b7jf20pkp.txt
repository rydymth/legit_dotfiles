    54	    /// <summary>
    55	    /// SystemName column widths are inconsistent across this schema (20 to 128) and 2014
    56	    /// does not name the column in a truncation error, so cap it here.
    57	    /// </summary>
    58	    private static string HostNameCapped()
    59	    {
    60	        string name = null;
    61	        try { name = System.Net.Dns.GetHostName(); } catch { /* not fatal */ }
    62	        if (string.IsNullOrEmpty(name)) return string.Empty;
    63	        return name.Length > 50 ? name.Substring(0, 50) : name;
    64	    }

    86	    /// <summary>
    87	    /// Fills the employee picker and returns how many employees are eligible.
    88	    ///
    89	    /// The count matters: an F&amp;F case can only be raised for someone who has actually
    90	    /// resigned or left, and in this database six of the eleven companies have nobody in
    91	    /// that state. An empty dropdown under a "pick the employee" instruction, with no
    92	    /// reason given, is indistinguishable from a broken screen - so the caller uses the
    93	    /// count to explain itself.
    94	    /// </summary>
    95	    private int LoadEligibleEmployees()
    96	    {
    97	        try
    98	        {
    99	            using (SqlConnection conn = CreateConnection())
   100	            {
   101	                conn.Open();
   102	                DataTable dt = FNFCaseService.EligibleEmployees(conn, null, CompanyId);
   103	                cmbEmployee.DataSource = dt;
   104	                cmbEmployee.ValueType = typeof(long);
   105	                cmbEmployee.DataBind();
   106	                return dt.Rows.Count;
   107	            }
   108	        }
   109	        catch (Exception ex)
   110	        {
   111	            ErrorLogHelper.Log(ex, "FNFSettlement.LoadEligible");
   112	            return -1;      // unknown - do not claim there is nobody
   113	        }
   114	    }

   165	    private DataTable LoadCases()
   166	    {
   167	        // An Admin / HR Admin sees every case in the company; anyone else sees only the
   168	        // cases they are named on. Scoped in the single query the grid runs, so the drill
   169	        // and the toolbar cannot widen it.
   170	        string scope = RoleHelper.IsAdmin(Session["UserRole"])
   171	            ? string.Empty
   172	            : @" AND (F.HRExecutiveUserMasterID = @UserID
   173	                   OR F.PayrollExecutiveUserMasterID = @UserID
   174	                   OR F.FinanceExecutiveUserMasterID = @UserID
   175	                   OR F.ApproverUserMasterID = @UserID
   176	                   OR F.EntryUserMasterID = @UserID) ";
   177	
   178	        string sql = @"
   179	            SELECT F.FNFSettlementID, F.CaseNo, F.CaseDate,
   180	                   LTRIM(ISNULL(' ' + NULLIF(LTRIM(RTRIM(EM.FirstName)),''), '') + ISNULL(' ' + NULLIF(LTRIM(RTRIM(EM.FatherName)),''), '') + ISNULL(' ' + NULLIF(LTRIM(RTRIM(EM.LastName)),''), '')) AS EmployeeName,
   181	                   ISNULL(EM.Code, '')      AS EmployeeCode,
   182	                   F.LastWorkingDate, F.CasePriority, F.WorkflowStatus,
   183	                   CASE WHEN ISNULL(F.IsClearanceReady, 0) = 1 THEN 'Ready'
   184	                        ELSE ISNULL(F.ClearanceBlockReason, 'Not checked') END AS ClearanceText
   185	            FROM   dbo.HR_FNFSettlement F
   186	            LEFT   JOIN dbo.HR_EmployeeMaster EM ON EM.EmployeeMasterID = F.EmployeeMasterID
   187	            WHERE  F.Status = 1
   188	              AND  F.CompanyMasterID = @CompanyID " + scope + @"
   189	            ORDER  BY F.FNFSettlementID DESC";
   190	
   191	        var table = new DataTable();
   192	        try
   193	        {
   194	            using (SqlConnection conn = CreateConnection())
   195	            using (var cmd = new SqlCommand(sql, conn))
   196	            {
   197	                cmd.Parameters.AddWithValue("@CompanyID", CompanyId);
   198	                cmd.Parameters.AddWithValue("@UserID", UserId);
   199	                using (var da = new SqlDataAdapter(cmd)) da.Fill(table);
   200	            }
   201	        }
   202	        catch (Exception ex)
   203	        {
   204	            ErrorLogHelper.Log(ex, "FNFSettlement.LoadCases");
   205	        }
   206	        return table;
   207	    }

   239	    protected void FnfPanel_Callback(object sender, CallbackEventArgsBase e)
   240	    {
   241	        string[] parts = (e.Parameter ?? string.Empty).Split('|');
   242	        if (parts.Length < 2) return;
   243	
   244	        string action = parts[0];
   245	        long id;
   246	        long.TryParse(parts[1], out id);
   247	
   248	        try
   249	        {
   250	            switch (action)
   251	            {
   252	                case "New":     HandleNew();          break;
   253	                case "Create":  HandleCreate();       break;
   254	                case "Open":    HandleOpen(id);       break;
   255	                case "Save":      HandleSave();      break;
   256	                case "Recheck":   HandleRecheck();   break;
   257	                case "Calculate": HandleCalculate(); break;
   258	                case "Submit":    HandleSubmit();    break;
   259	            }
   260	        }
   261	        catch (Exception ex)
   262	        {
   263	            ErrorLogHelper.Log(ex, "FNFSettlement." + action);
   264	            FnfPanel.JSProperties["cpMsg"] = ErrorLogHelper.FriendlyMessage;
   265	        }
   266	    }

   328	    private void HandleCreate()
   329	    {
   330	        if (cmbEmployee.Value == null)
   331	        {
   332	            FnfPanel.JSProperties["cpMsg"] = "Pick the employee to settle first.";
   333	            return;
   334	        }
   335	
   336	        long employeeId = Convert.ToInt64(cmbEmployee.Value);
   337	        int windowDays = ComplianceWindowDaysForCompany();
   338	        long newId;
   339	
   340	        using (SqlConnection conn = CreateConnection())
   341	        {
   342	            conn.Open();
   343	            // One transaction for the whole thing: the header, the nine checklist rows, the
   344	            // auto-answers, the discovered recoveries and the clearance verdict. A case must
   345	            // never exist with a half-built checklist.
   346	            using (SqlTransaction tx = conn.BeginTransaction())
   347	            {
   348	                try
   349	                {
   350	                    newId = FNFCaseService.CreateCase(conn, tx, employeeId, CompanyId,
   351	                                                      UserId, HostNameCapped(), windowDays);
   352	                    tx.Commit();
   353	                }
   354	                catch
   355	                {
   356	                    try { tx.Rollback(); } catch { /* connection already gone */ }
   357	                    throw;
   358	                }
   359	            }
   360	        }
   361	
   362	        LoadCase(newId);
   363	        FnfPanel.JSProperties["cpMsg"] = "F&F case " + txtCaseNo.Text + " created.";
   364	    }

   381	    private void LoadCase(long id)
   382	    {
   383	        const string sql = @"
   384	            SELECT F.*,
   385	                   LTRIM(ISNULL(' ' + NULLIF(LTRIM(RTRIM(EM.FirstName)),''), '') + ISNULL(' ' + NULLIF(LTRIM(RTRIM(EM.FatherName)),''), '') + ISNULL(' ' + NULLIF(LTRIM(RTRIM(EM.LastName)),''), '')) AS EmployeeName,
   386	                   ISNULL(EM.Code, '')          AS EmployeeCode,
   387	                   -- These two tables name their caption columns DepartmentName /
   388	                   -- DesignationName, not Name. Getting that wrong made both Create and
   389	                   -- Open throw Invalid column name Name, so the case was created but its
   390	                   -- snapshot never loaded and the form sat on its placeholder text.
   391	                   ISNULL(D.DepartmentName, '')  AS DepartmentName,
   392	                   ISNULL(G.DesignationName, '') AS DesignationName,
   393	                   LTRIM(ISNULL(' ' + NULLIF(LTRIM(RTRIM(MG.FirstName)),''), '') + ISNULL(' ' + NULLIF(LTRIM(RTRIM(MG.LastName)),''), '')) AS ManagerName,
   394	                   EBS.BasicAmount, EBS.GrossAmount, EBS.CTCAmount
   395	            FROM   dbo.HR_FNFSettlement F
   396	            LEFT   JOIN dbo.HR_EmployeeMaster   EM ON EM.EmployeeMasterID   = F.EmployeeMasterID
   397	            LEFT   JOIN dbo.HR_DepartmentMaster D  ON D.DepartmentMasterID  = F.DepartmentMasterID
   398	            LEFT   JOIN dbo.HR_DesignationMaster G ON G.DesignationMasterID = F.DesignationMasterID
   399	            LEFT   JOIN dbo.HR_EmployeeMaster   MG ON MG.EmployeeMasterID   = F.ReportingPersonID
   400	            OUTER  APPLY (SELECT TOP 1 B.BasicAmount, B.GrossAmount, B.CTCAmount
   401	                          FROM   dbo.HR_EmployeeBasicSalary B
   402	                          WHERE  B.EmployeeMasterID = F.EmployeeMasterID
   403	                          ORDER  BY B.FromDate DESC) EBS
   404	            WHERE  F.FNFSettlementID = @Id";
   405	
   406	        DataRow row = null;
   407	        using (SqlConnection conn = CreateConnection())
   408	        using (var cmd = new SqlCommand(sql, conn))
   409	        {
   410	            cmd.Parameters.AddWithValue("@Id", id);
   411	            var dt = new DataTable();
   412	            using (var da = new SqlDataAdapter(cmd)) da.Fill(dt);
   413	            if (dt.Rows.Count > 0) row = dt.Rows[0];
   414	        }
   415	
   416	        if (row == null)
   417	        {
   418	            FnfPanel.JSProperties["cpMsg"] = "That F&F case no longer exists.";
   419	            FnfPanel.JSProperties["cpReloadList"] = true;
   420	            return;
   421	        }
   422	
   423	        CurrentCaseId = id;
   424	        CurrentStatus = Convert.ToString(row["WorkflowStatus"]);
   425	
   426	        txtCaseNo.Text = Convert.ToString(row["CaseNo"]);
   427	        txtStatus.Text = CurrentStatus;
   428	        dtCaseDate.Value = row["CaseDate"];
   429	        cmbPriority.Value = Convert.ToString(row["CasePriority"]);
   430	
   431	        // The employee cannot be changed once the case exists - the whole snapshot was
   432	        // taken from them. Shown for reference, locked against editing.
   433	        cmbEmployee.Value = Convert.ToInt64(row["EmployeeMasterID"]);
   434	        cmbEmployee.Enabled = false;
   435	
   436	        dtPlannedCalc.Value   = NullIfDbNull(row["PlannedCalcStartDate"]);
   437	        dtPlannedSettle.Value = NullIfDbNull(row["PlannedSettlementDate"]);
   438	        dtActualCalc.Value    = NullIfDbNull(row["ActualCalcStartDate"]);
   439	        spnTat.Value          = NullIfDbNull(row["ApprovalTurnaroundDays"]);
   440	        cmbHrExec.Value       = NullIfDbNull(row["HRExecutiveUserMasterID"]);
   441	        cmbPayrollExec.Value  = NullIfDbNull(row["PayrollExecutiveUserMasterID"]);
   442	        cmbFinanceExec.Value  = NullIfDbNull(row["FinanceExecutiveUserMasterID"]);
   443	        cmbApprover.Value     = NullIfDbNull(row["ApproverUserMasterID"]);
   444	        memRemarks.Text       = Convert.ToString(row["Remarks"]);
   445	
   446	        litSnapshot.Text = BuildSnapshotHtml(row);
   447	        litBanner.Text = BuildBannerHtml(row);
   448	
   449	        ReadinessGrid.DataBind();
   450	        RecoveryGrid.DataBind();
   451	        SettleGrid.DataBind();
   452	        VerifyGrid.DataBind();
   453	        HandoverGrid.DataBind();
   454	        RenderTotals();
   455	        RenderSettlementNotes(row);
   456	
   457	        bool editable = FNFCaseService.CanEdit(CurrentStatus);
   458	        FnfPanel.JSProperties["cpCreating"] = "0";
   459	        FnfPanel.JSProperties["cpLocked"] = editable ? "0" : "1";
   460	    }

   489	    private string BuildSnapshotHtml(DataRow row)
   490	    {
   491	        var sb = new System.Text.StringBuilder();
   492	        sb.Append("<table class=\"fnf-kv\" style=\"border-collapse:collapse\"><tr>");
   493	        sb.Append(Cell("Employee", row["EmployeeName"]));
   494	        sb.Append(Cell("Code", row["EmployeeCode"]));
   495	        sb.Append(Cell("Department", row["DepartmentName"]));
   496	        sb.Append(Cell("Designation", row["DesignationName"]));
   497	        sb.Append(Cell("Manager", row["ManagerName"]));
   498	        sb.Append("</tr><tr>");

   500	        sb.Append(DateCell("Resignation Date", row["ResignDate"]));
   501	        sb.Append(DateCell("Last Working Day", row["LastWorkingDate"]));

   503	        sb.Append(Cell("Notice Served (days)", NoticeServedDays(row)));
   504	        sb.Append("</tr><tr>");
   505	        sb.Append(MoneyCell("Basic", row["BasicAmount"]));
   506	        sb.Append(MoneyCell("Gross", row["GrossAmount"]));
   507	        sb.Append(MoneyCell("CTC", row["CTCAmount"]));
   508	        sb.Append(DateCell("Compliance Window Ends", row["ComplianceWindowEnd"]));
   509	        sb.Append("</tr></table>");
   510	        return sb.ToString();
   511	    }

   513	    /// <summary>
   514	    /// Days actually served between resignation and the last working day. Derived rather
   515	    /// than stored: both dates are on the case, so a stored copy could only ever disagree
   516	    /// with them.
   517	    /// </summary>
   518	    private static object NoticeServedDays(DataRow row)
   519	    {
   520	        if (row["ResignDate"] == DBNull.Value || row["LastWorkingDate"] == DBNull.Value)
   521	            return null;
   522	
   523	        DateTime from = Convert.ToDateTime(row["ResignDate"]);
   524	        DateTime to = Convert.ToDateTime(row["LastWorkingDate"]);
   525	        int days = (int)(to.Date - from.Date).TotalDays;
   526	        return days < 0 ? 0 : days;
   527	    }

   529	    private string BuildBannerHtml(DataRow row)
   530	    {
   531	        bool editable = FNFCaseService.CanEdit(Convert.ToString(row["WorkflowStatus"]));
   532	        if (!editable)
   533	        {

   535	                   HttpUtility.HtmlEncode(Convert.ToString(row["WorkflowStatus"])) +
   536	                   " and is shown read-only.</div>";
   537	        }
   538	
   539	        bool ready = row["IsClearanceReady"] != DBNull.Value && Convert.ToBoolean(row["IsClearanceReady"]);
   540	        if (ready)
   541	        {

   543	                   "calculated for this case.</div>";

   553	    /// <summary>
   554	    /// Re-runs the calculation WITHOUT saving, purely to collect its notes for the screen.
   555	    ///
   556	    /// The notes are the difference between "gratuity is zero because service is too short"
   557	    /// and "gratuity is zero because nobody switched it on" — a settlement that quietly
   558	    /// skipped a component looks identical to one where it is genuinely nil, and that is the
   559	    /// single most misleading thing this screen could do.
   560	    /// </summary>
   561	    private void RenderSettlementNotes(DataRow head)
   562	    {
   563	        litSettleNotes.Text = string.Empty;
   564	        if (CurrentCaseId <= 0) return;
   565	
   566	        try
   567	        {
   568	            List<string> notes;
   569	            using (SqlConnection conn = CreateConnection())
   570	            {
   571	                conn.Open();
   572	                using (SqlTransaction tx = conn.BeginTransaction())
   573	                {
   574	                    try
   575	                    {
   576	                        notes = FNFSettlementService.Calculate(conn, tx, CurrentCaseId, UserId, false).Notes;
   577	                    }
   578	                    finally
   579	                    {
   580	                        // Read-only by construction: persist = false, and the transaction is
   581	                        // always rolled back so nothing here can write.
   582	                        try { tx.Rollback(); } catch { }
   583	                    }
   584	                }
   585	            }
   586	
   587	            if (notes.Count == 0) return;
   588	
   589	            var sb = new System.Text.StringBuilder();
   590	            sb.Append("<div class=\"fnf-banner locked\"><b>Worth knowing about this settlement</b><ul style=\"margin:6px 0 0 18px;font-weight:400\">");
   591	            foreach (string note in notes)
   592	                sb.Append("<li>").Append(HttpUtility.HtmlEncode(note)).Append("</li>");
   593	            sb.Append("</ul></div>");
   594	            litSettleNotes.Text = sb.ToString();
   595	        }
   596	        catch (Exception ex)
   597	        {
   598	            // A case with no config or no salary record must still open.
   599	            ErrorLogHelper.Log(ex, "FNFSettlement.Notes");
   600	        }
   601	    }

   670	    /// <summary>
   671	    /// Re-reads attendance, leave and the ledgers, rescans for recoveries and recomputes the
   672	    /// clearance verdict. Needed because all three move underneath the case: Confirm Pay may
   673	    /// run after the case was raised, and an asset may come back later.
   674	    /// </summary>
   675	    private void HandleRecheck()
   676	    {
   677	        if (CurrentCaseId <= 0) return;
   678	
   679	        int found;
   680	        using (SqlConnection conn = CreateConnection())
   681	        {
   682	            conn.Open();
   683	            using (SqlTransaction tx = conn.BeginTransaction())
   684	            {
   685	                try
   686	                {
   687	                    FNFCaseService.SeedReadiness(conn, tx, CurrentCaseId);
   688	                    FNFCaseService.AutoAnswerReadiness(conn, tx, CurrentCaseId);
   689	                    found = FNFCaseService.DiscoverRecoveries(conn, tx, CurrentCaseId, UserId, HostNameCapped());
   690	                    FNFCaseService.EvaluateClearance(conn, tx, CurrentCaseId);
   691	                    tx.Commit();
   692	                }
   693	                catch
   694	                {
   695	                    try { tx.Rollback(); } catch { /* connection already gone */ }
   696	                    throw;
   697	                }
   698	            }
   699	        }
   700	
   701	        LoadCase(CurrentCaseId);
   702	        FnfPanel.JSProperties["cpMsg"] = found > 0
   703	            ? "Readiness re-checked. " + found + " new recovery row(s) found."
   704	            : "Readiness re-checked. No new recoveries.";
   705	    }

   709	    /// <summary>
   710	    /// Works out the settlement and seeds the verification and handover checklists.
   711	    ///
   712	    /// Refuses while clearance is not ready — that is the whole point of the readiness
   713	    /// checklist: calculating on incomplete data produces a figure somebody will believe.
   714	    /// </summary>
   715	    private void HandleCalculate()
   716	    {
   717	        if (CurrentCaseId <= 0)
   718	        {
   719	            FnfPanel.JSProperties["cpMsg"] = "Open a case first.";
   720	            return;
   721	        }
   722	
   723	        if (!FNFCaseService.CanEdit(CurrentStatus))
   724	        {
   725	            FnfPanel.JSProperties["cpMsg"] = "This case is " + CurrentStatus + " and can no longer be changed.";
   726	            return;
   727	        }
   728	
   729	        FNFSettlementService.Result result;
   730	
   731	        using (SqlConnection conn = CreateConnection())
   732	        {
   733	            conn.Open();
   734	
   735	            if (!IsClearanceReady(conn, null))
   736	            {
   737	                FnfPanel.JSProperties["cpMsg"] =
   738	                    "Clearance is not ready yet, so the settlement was not calculated. "
   739	                  + "Complete the Clearance Readiness tab first.";
   740	                return;
   741	            }
   742	
   743	            using (SqlTransaction tx = conn.BeginTransaction())
   744	            {
   745	                try
   746	                {
   747	                    result = FNFSettlementService.Calculate(conn, tx, CurrentCaseId, UserId, true);
   748	                    // Seeded from the service so the approval screen and the tests do it
   749	                    // identically - this used to be a private copy on this page.
   750	                    FNFSettlementService.SeedVerificationAndHandover(conn, tx, CurrentCaseId, result);
   751	                    tx.Commit();
   752	                }
   753	                catch
   754	                {
   755	                    try { tx.Rollback(); } catch { /* connection already gone */ }
   756	                    throw;
   757	                }
   758	            }
   759	        }
   760	
   761	        LoadCase(CurrentCaseId);
   762	
   763	        FnfPanel.JSProperties["cpMsg"] = "Calculated. Net payable "
   764	            + result.NetPayable.ToString("n2")
   765	            + (result.Notes.Count > 0 ? " - see the notes on the Settlement tab." : ".");
   766	    }

   768	    private bool IsClearanceReady(SqlConnection conn, SqlTransaction tx)
   769	    {
   770	        using (var cmd = new SqlCommand(
   771	            "SELECT ISNULL(CAST(IsClearanceReady AS int), 0) FROM dbo.HR_FNFSettlement WHERE FNFSettlementID = @Id",
   772	            conn, tx))
   773	        {
   774	            cmd.Parameters.AddWithValue("@Id", CurrentCaseId);
   775	            object value = cmd.ExecuteScalar();
   776	            return value != null && value != DBNull.Value && Convert.ToInt32(value) == 1;
   777	        }
   778	    }
